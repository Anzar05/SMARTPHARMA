*---------------------------------------------------------------------*
* CUSTOMER DISPLAY
*---------------------------------------------------------------------*

FORM customer_display.

  CLEAR gt_customer.

  SELECT *
    FROM zsp_customer
    INTO TABLE @gt_customer.

  IF gt_customer IS INITIAL.

    MESSAGE 'No customer data found.' TYPE 'S'.

    RETURN.

  ENDIF.

  TRY.

      DATA lo_customer TYPE REF TO cl_salv_table.

      cl_salv_table=>factory(
        IMPORTING
          r_salv_table = lo_customer
        CHANGING
          t_table      = gt_customer ).

      lo_customer->get_functions( )->set_all( abap_true ).

      lo_customer->get_columns( )->set_optimize( abap_true ).

      PERFORM set_alv_headers USING lo_customer.

      lo_customer->get_display_settings(
        )->set_striped_pattern( abap_true ).

      lo_customer->get_display_settings(
        )->set_list_header(
        'SMARTPHARMA CUSTOMER MASTER' ).

      lo_customer->display( ).

    CATCH cx_salv_msg INTO DATA(lx_customer).

      MESSAGE lx_customer->get_text( ) TYPE 'S'.

  ENDTRY.

ENDFORM.

*---------------------------------------------------------------------*
* CUSTOMER HISTORY
*---------------------------------------------------------------------*

FORM customer_history.

  CLEAR gt_chis.

  SELECT
    c~customer_id,
    c~customer_name,
    c~mobile,
    h~sales_id,
    h~sales_date,
    i~matnr,
    i~batch_no,
    i~quantity,
    i~unit,
    i~sales_price,
    i~total_amount AS item_amount
    FROM zsp_customer AS c
    INNER JOIN zsp_sales_hdr AS h
      ON c~customer_id = h~customer_id
    INNER JOIN zsp_sales_item AS i
      ON h~sales_id = i~sales_id
    INTO CORRESPONDING FIELDS OF TABLE @gt_chis.

  IF gt_chis IS INITIAL.

    MESSAGE 'No customer purchase history found.' TYPE 'S'.

    RETURN.

  ENDIF.

  TRY.

      DATA lo_chis TYPE REF TO cl_salv_table.

      cl_salv_table=>factory(
        IMPORTING
          r_salv_table = lo_chis
        CHANGING
          t_table      = gt_chis ).

      lo_chis->get_functions( )->set_all( abap_true ).

      lo_chis->get_columns( )->set_optimize( abap_true ).

      PERFORM set_alv_headers USING lo_chis.

      lo_chis->get_display_settings(
        )->set_striped_pattern( abap_true ).

      lo_chis->get_display_settings(
        )->set_list_header(
        'SMARTPHARMA CUSTOMER PURCHASE HISTORY' ).

      lo_chis->display( ).

    CATCH cx_salv_msg INTO DATA(lx_chis).

      MESSAGE lx_chis->get_text( ) TYPE 'S'.

  ENDTRY.

ENDFORM.

*---------------------------------------------------------------------*
* CUSTOMER CREATE
*---------------------------------------------------------------------*

FORM customer_create.

  DATA:
    lv_count TYPE i,
    lv_id    TYPE char10.

  IF p_cnam IS INITIAL.

    MESSAGE 'Enter customer name.' TYPE 'S'.

    RETURN.

  ENDIF.

  IF p_cmob IS INITIAL.

    MESSAGE 'Enter customer mobile number.' TYPE 'S'.

    RETURN.

  ENDIF.

  SELECT SINGLE *
    FROM zsp_customer
    INTO @gs_customer
    WHERE mobile = @p_cmob.

  IF sy-subrc = 0.

    MESSAGE 'Customer with this mobile number already exists.' TYPE 'S'.

    RETURN.

  ENDIF.

  SELECT COUNT( * )
    FROM zsp_customer
    INTO @lv_count.

  lv_count = lv_count + 1.

  lv_id = |C{ lv_count }|.

  CLEAR gs_customer.

  gs_customer-mandt = sy-mandt.
  gs_customer-customer_id = lv_id.
  gs_customer-customer_name = p_cnam.
  gs_customer-contact_person = p_ccon.
  gs_customer-mobile = p_cmob.
  gs_customer-gst_number = p_cgst.
  gs_customer-address = p_cadd.
  gs_customer-customer_type = p_ctyp.
  gs_customer-created_on = sy-datum.

  INSERT zsp_customer FROM @gs_customer.

  IF sy-subrc = 0.

    COMMIT WORK AND WAIT.

    MESSAGE 'Customer created successfully.' TYPE 'S'.

  ELSE.

    ROLLBACK WORK.

    MESSAGE 'Customer creation failed.' TYPE 'S'.

  ENDIF.

ENDFORM.

*---------------------------------------------------------------------*
* PURCHASE CREATE
*---------------------------------------------------------------------*

FORM purchase_create.

  DATA:
    lv_count TYPE i,
    lv_pid   TYPE char12,
    lv_total TYPE p LENGTH 8 DECIMALS 2.

  IF p_pvid IS INITIAL.

    MESSAGE 'Enter vendor ID.' TYPE 'S'.

    RETURN.

  ENDIF.

  IF p_pmat IS INITIAL.

    MESSAGE 'Enter material number.' TYPE 'S'.

    RETURN.

  ENDIF.

  IF p_pbat IS INITIAL.

    MESSAGE 'Enter batch number.' TYPE 'S'.

    RETURN.

  ENDIF.

  IF p_pqty <= 0.

    MESSAGE 'Enter valid purchase quantity.' TYPE 'S'.

    RETURN.

  ENDIF.

  IF p_pprc <= 0.

    MESSAGE 'Enter valid purchase price.' TYPE 'S'.

    RETURN.

  ENDIF.

  IF p_pspr <= 0.

    MESSAGE 'Enter valid sales price.' TYPE 'S'.

    RETURN.

  ENDIF.

  SELECT SINGLE *
    FROM zsp_vendor
    INTO @gs_vendor
    WHERE vendor_id = @p_pvid.

  IF sy-subrc <> 0.

    MESSAGE 'Vendor ID does not exist.' TYPE 'S'.

    RETURN.

  ENDIF.

  lv_total = p_pqty * p_pprc.

  SELECT COUNT( * )
    FROM zsp_purchase_hdr
    INTO @lv_count.

  lv_count = lv_count + 1.

  lv_pid = |P{ lv_count }|.

  CLEAR gs_phdr.

  gs_phdr-mandt = sy-mandt.
  gs_phdr-purchase_id = lv_pid.
  gs_phdr-vendor_id = p_pvid.
  gs_phdr-purchase_date = sy-datum.
  gs_phdr-currency = p_pcur.
  gs_phdr-total_amount = lv_total.
  gs_phdr-created_on = sy-datum.

  INSERT zsp_purchase_hdr FROM @gs_phdr.

  IF sy-subrc <> 0.

    ROLLBACK WORK.

    MESSAGE 'Purchase header creation failed.' TYPE 'S'.

    RETURN.

  ENDIF.

  CLEAR gs_pitem.

  gs_pitem-mandt = sy-mandt.
  gs_pitem-purchase_id = lv_pid.
  gs_pitem-item_no = '1'.
  gs_pitem-matnr = p_pmat.
  gs_pitem-batch_no = p_pbat.
  gs_pitem-quantity = p_pqty.
  gs_pitem-unit = p_puni.
  gs_pitem-purchase_price = p_pprc.
  gs_pitem-total_amount = lv_total.

  INSERT zsp_purch_item FROM @gs_pitem.

  IF sy-subrc <> 0.

    ROLLBACK WORK.

    MESSAGE 'Purchase item creation failed.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* UPDATE OR CREATE STOCK
*---------------------------------------------------------------------*

  SELECT SINGLE *
    FROM zsp_inv_stock
    INTO @gs_stockdb
    WHERE matnr = @p_pmat
      AND batch_no = @p_pbat.

  IF sy-subrc = 0.

    gs_stockdb-quantity =
      gs_stockdb-quantity + p_pqty.

    gs_stockdb-purchase_price = p_pprc.
    gs_stockdb-sales_price = p_pspr.
    gs_stockdb-unit = p_puni.
    gs_stockdb-updated_on = sy-datum.

    MODIFY zsp_inv_stock FROM @gs_stockdb.

  ELSE.

    CLEAR gs_stockdb.

    gs_stockdb-mandt = sy-mandt.
    gs_stockdb-matnr = p_pmat.
    gs_stockdb-batch_no = p_pbat.
    gs_stockdb-quantity = p_pqty.
    gs_stockdb-unit = p_puni.
    gs_stockdb-purchase_price = p_pprc.
    gs_stockdb-sales_price = p_pspr.
    gs_stockdb-updated_on = sy-datum.

    INSERT zsp_inv_stock FROM @gs_stockdb.

  ENDIF.

  IF sy-subrc = 0.

    COMMIT WORK AND WAIT.

    MESSAGE 'Purchase created and stock updated successfully.'
      TYPE 'S'.

  ELSE.

    ROLLBACK WORK.

    MESSAGE 'Stock update failed.' TYPE 'S'.

  ENDIF.

ENDFORM.

*---------------------------------------------------------------------*
* BILLING CREATE
*---------------------------------------------------------------------*

FORM billing_create.

  DATA:
    ls_customer TYPE zsp_customer,
    ls_stock1   TYPE zsp_inv_stock,
    ls_stock2   TYPE zsp_inv_stock,
    ls_stock3   TYPE zsp_inv_stock,
    lv_count    TYPE i,
    lv_sid      TYPE char12,
    lv_total    TYPE p LENGTH 8 DECIMALS 2,
    lv_amt      TYPE p LENGTH 8 DECIMALS 2.

*---------------------------------------------------------------------*
* MOBILE NUMBER
*---------------------------------------------------------------------*

  IF p_bmob IS INITIAL.

    MESSAGE 'Enter customer mobile number.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* CUSTOMER SEARCH
*---------------------------------------------------------------------*

  CLEAR ls_customer.

  SELECT SINGLE *
    FROM zsp_customer
    INTO @ls_customer
    WHERE mobile = @p_bmob.

  IF sy-subrc = 0.

*---------------------------------------------------------------------*
* EXISTING CUSTOMER
*---------------------------------------------------------------------*

    MESSAGE 'Regular customer found. Existing customer will be used.'
      TYPE 'S'.

  ELSE.

*---------------------------------------------------------------------*
* NEW CUSTOMER
*---------------------------------------------------------------------*

    IF p_bnam IS INITIAL.

      MESSAGE 'New customer: enter customer name.' TYPE 'S'.

      RETURN.

    ENDIF.

    SELECT COUNT( * )
      FROM zsp_customer
      INTO @lv_count.

    lv_count = lv_count + 1.

    CLEAR ls_customer.

    ls_customer-mandt = sy-mandt.

    ls_customer-customer_id = |C{ lv_count }|.

    ls_customer-customer_name = p_bnam.
    ls_customer-contact_person = p_bcon.
    ls_customer-mobile = p_bmob.
    ls_customer-gst_number = p_bgst.
    ls_customer-address = p_badd.
    ls_customer-customer_type = p_btyp.
    ls_customer-created_on = sy-datum.

    INSERT zsp_customer FROM @ls_customer.

    IF sy-subrc <> 0.

      ROLLBACK WORK.

      MESSAGE 'New customer creation failed.' TYPE 'S'.

      RETURN.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* MEDICINE 1
*---------------------------------------------------------------------*

  IF p_bm1 IS INITIAL.

    MESSAGE 'Enter at least one medicine material number.' TYPE 'S'.

    ROLLBACK WORK.

    RETURN.

  ENDIF.

  IF p_bb1 IS INITIAL.

    MESSAGE 'Enter batch number for first medicine.' TYPE 'S'.

    ROLLBACK WORK.

    RETURN.

  ENDIF.

  IF p_bq1 <= 0.

    MESSAGE 'Enter valid quantity for first medicine.' TYPE 'S'.

    ROLLBACK WORK.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* STOCK CHECK 1
*---------------------------------------------------------------------*

  SELECT SINGLE *
    FROM zsp_inv_stock
    INTO @ls_stock1
    WHERE matnr = @p_bm1
      AND batch_no = @p_bb1.

  IF sy-subrc <> 0.

    ROLLBACK WORK.

    MESSAGE 'First medicine and batch were not found in stock.'
      TYPE 'S'.

    RETURN.

  ENDIF.

  IF ls_stock1-quantity < p_bq1.

    ROLLBACK WORK.

    MESSAGE 'Insufficient stock for first medicine.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* MEDICINE 2
*---------------------------------------------------------------------*

  IF p_bm2 IS NOT INITIAL.

    IF p_bb2 IS INITIAL.

      ROLLBACK WORK.

      MESSAGE 'Enter batch number for second medicine.' TYPE 'S'.

      RETURN.

    ENDIF.

    IF p_bq2 <= 0.

      ROLLBACK WORK.

      MESSAGE 'Enter valid quantity for second medicine.' TYPE 'S'.

      RETURN.

    ENDIF.

    SELECT SINGLE *
      FROM zsp_inv_stock
      INTO @ls_stock2
      WHERE matnr = @p_bm2
        AND batch_no = @p_bb2.

    IF sy-subrc <> 0.

      ROLLBACK WORK.

      MESSAGE 'Second medicine and batch were not found in stock.'
        TYPE 'S'.

      RETURN.

    ENDIF.

    IF ls_stock2-quantity < p_bq2.

      ROLLBACK WORK.

      MESSAGE 'Insufficient stock for second medicine.' TYPE 'S'.

      RETURN.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* MEDICINE 3
*---------------------------------------------------------------------*

  IF p_bm3 IS NOT INITIAL.

    IF p_bb3 IS INITIAL.

      ROLLBACK WORK.

      MESSAGE 'Enter batch number for third medicine.' TYPE 'S'.

      RETURN.

    ENDIF.

    IF p_bq3 <= 0.

      ROLLBACK WORK.

      MESSAGE 'Enter valid quantity for third medicine.' TYPE 'S'.

      RETURN.

    ENDIF.

    SELECT SINGLE *
      FROM zsp_inv_stock
      INTO @ls_stock3
      WHERE matnr = @p_bm3
        AND batch_no = @p_bb3.

    IF sy-subrc <> 0.

      ROLLBACK WORK.

      MESSAGE 'Third medicine and batch were not found in stock.'
        TYPE 'S'.

      RETURN.

    ENDIF.

    IF ls_stock3-quantity < p_bq3.

      ROLLBACK WORK.

      MESSAGE 'Insufficient stock for third medicine.' TYPE 'S'.

      RETURN.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* BILL TOTAL
*---------------------------------------------------------------------*

  CLEAR lv_total.

  lv_amt = p_bq1 * ls_stock1-sales_price.

  lv_total = lv_total + lv_amt.

  IF p_bm2 IS NOT INITIAL.

    lv_amt = p_bq2 * ls_stock2-sales_price.

    lv_total = lv_total + lv_amt.

  ENDIF.

  IF p_bm3 IS NOT INITIAL.

    lv_amt = p_bq3 * ls_stock3-sales_price.

    lv_total = lv_total + lv_amt.

  ENDIF.

*---------------------------------------------------------------------*
* BILL NUMBER
*---------------------------------------------------------------------*

  SELECT COUNT( * )
    FROM zsp_sales_hdr
    INTO @lv_count.

  lv_count = lv_count + 1.

  lv_sid = |S{ lv_count }|.

*---------------------------------------------------------------------*
* SALES HEADER
*---------------------------------------------------------------------*

  CLEAR gs_shdr.

  gs_shdr-mandt = sy-mandt.
  gs_shdr-sales_id = lv_sid.
  gs_shdr-customer_id = ls_customer-customer_id.
  gs_shdr-sales_date = sy-datum.
  gs_shdr-currency = p_bcur.
  gs_shdr-total_amount = lv_total.
  gs_shdr-created_on = sy-datum.

  INSERT zsp_sales_hdr FROM @gs_shdr.

  IF sy-subrc <> 0.

    ROLLBACK WORK.

    MESSAGE 'Bill header creation failed.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* BILL ITEM 1
*---------------------------------------------------------------------*

  CLEAR gs_sitem.

  gs_sitem-mandt = sy-mandt.
  gs_sitem-sales_id = lv_sid.
  gs_sitem-item_no = '1'.
  gs_sitem-matnr = p_bm1.
  gs_sitem-batch_no = p_bb1.
  gs_sitem-quantity = p_bq1.
  gs_sitem-unit = ls_stock1-unit.
  gs_sitem-sales_price = ls_stock1-sales_price.
  gs_sitem-total_amount =
    p_bq1 * ls_stock1-sales_price.

  INSERT zsp_sales_item FROM @gs_sitem.

  IF sy-subrc <> 0.

    ROLLBACK WORK.

    MESSAGE 'First bill item could not be saved.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* REDUCE STOCK 1
*---------------------------------------------------------------------*

  ls_stock1-quantity =
    ls_stock1-quantity - p_bq1.

  ls_stock1-updated_on = sy-datum.

  MODIFY zsp_inv_stock FROM @ls_stock1.

  IF sy-subrc <> 0.

    ROLLBACK WORK.

    MESSAGE 'First stock update failed.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* BILL ITEM 2
*---------------------------------------------------------------------*

  IF p_bm2 IS NOT INITIAL.

    CLEAR gs_sitem.

    gs_sitem-mandt = sy-mandt.
    gs_sitem-sales_id = lv_sid.
    gs_sitem-item_no = '2'.
    gs_sitem-matnr = p_bm2.
    gs_sitem-batch_no = p_bb2.
    gs_sitem-quantity = p_bq2.
    gs_sitem-unit = ls_stock2-unit.
    gs_sitem-sales_price = ls_stock2-sales_price.
    gs_sitem-total_amount =
      p_bq2 * ls_stock2-sales_price.

    INSERT zsp_sales_item FROM @gs_sitem.

    IF sy-subrc <> 0.

      ROLLBACK WORK.

      MESSAGE 'Second bill item could not be saved.' TYPE 'S'.

      RETURN.

    ENDIF.

    ls_stock2-quantity =
      ls_stock2-quantity - p_bq2.

    ls_stock2-updated_on = sy-datum.

    MODIFY zsp_inv_stock FROM @ls_stock2.

    IF sy-subrc <> 0.

      ROLLBACK WORK.

      MESSAGE 'Second stock update failed.' TYPE 'S'.

      RETURN.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* BILL ITEM 3
*---------------------------------------------------------------------*

  IF p_bm3 IS NOT INITIAL.

    CLEAR gs_sitem.

    gs_sitem-mandt = sy-mandt.
    gs_sitem-sales_id = lv_sid.
    gs_sitem-item_no = '3'.
    gs_sitem-matnr = p_bm3.
    gs_sitem-batch_no = p_bb3.
    gs_sitem-quantity = p_bq3.
    gs_sitem-unit = ls_stock3-unit.
    gs_sitem-sales_price = ls_stock3-sales_price.
    gs_sitem-total_amount =
      p_bq3 * ls_stock3-sales_price.

    INSERT zsp_sales_item FROM @gs_sitem.

    IF sy-subrc <> 0.

      ROLLBACK WORK.

      MESSAGE 'Third bill item could not be saved.' TYPE 'S'.

      RETURN.

    ENDIF.

    ls_stock3-quantity =
      ls_stock3-quantity - p_bq3.

    ls_stock3-updated_on = sy-datum.

    MODIFY zsp_inv_stock FROM @ls_stock3.

    IF sy-subrc <> 0.

      ROLLBACK WORK.

      MESSAGE 'Third stock update failed.' TYPE 'S'.

      RETURN.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* SAVE EVERYTHING
*---------------------------------------------------------------------*

  COMMIT WORK AND WAIT.

*---------------------------------------------------------------------*
* BILL SUCCESS
*---------------------------------------------------------------------*

  MESSAGE |Bill { lv_sid } generated successfully. Total: { lv_total } { p_bcur }|
    TYPE 'S'.

ENDFORM.

*---------------------------------------------------------------------*
* STOCK DISPLAY
*---------------------------------------------------------------------*

FORM stock_display USING pv_mode TYPE char4.

  DATA:
    lv_near TYPE dats.

  CLEAR gt_stock.

  lv_near = sy-datum + 30.

  SELECT
    mandt,
    matnr,
    batch_no,
    quantity,
    unit,
    purchase_price,
    sales_price,
    expiry_date,
    updated_on
    FROM zsp_inv_stock
    INTO CORRESPONDING FIELDS OF TABLE @gt_stock.

  IF gt_stock IS INITIAL.

    MESSAGE 'No stock data found.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* STATUS CALCULATION
*---------------------------------------------------------------------*

  LOOP AT gt_stock ASSIGNING FIELD-SYMBOL(<fs_stock>).

    IF <fs_stock>-quantity <= 0.

      <fs_stock>-status = 'LOW STOCK'.

    ELSEIF <fs_stock>-expiry_date IS NOT INITIAL
       AND <fs_stock>-expiry_date < sy-datum.

      <fs_stock>-status = 'EXPIRED'.

    ELSEIF <fs_stock>-quantity <= 10.

      <fs_stock>-status = 'LOW STOCK'.

    ELSEIF <fs_stock>-expiry_date IS NOT INITIAL
       AND <fs_stock>-expiry_date <= lv_near.

      <fs_stock>-status = 'NEAR EXPIRY'.

    ELSE.

      <fs_stock>-status = 'AVAILABLE'.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* FILTER STOCK
*---------------------------------------------------------------------*

  IF pv_mode = 'LOW'.

    DELETE gt_stock
      WHERE status <> 'LOW STOCK'.

  ELSEIF pv_mode = 'NEAR'.

    DELETE gt_stock
      WHERE status <> 'NEAR EXPIRY'.

  ELSEIF pv_mode = 'EXP'.

    DELETE gt_stock
      WHERE status <> 'EXPIRED'.

  ENDIF.

  IF gt_stock IS INITIAL.

    MESSAGE 'No records found for selected stock category.' TYPE 'S'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* STOCK ALV
*---------------------------------------------------------------------*

  TRY.

      DATA lo_stock TYPE REF TO cl_salv_table.

      cl_salv_table=>factory(
        IMPORTING
          r_salv_table = lo_stock
        CHANGING
          t_table      = gt_stock ).

      lo_stock->get_functions( )->set_all( abap_true ).

      lo_stock->get_columns( )->set_optimize( abap_true ).

      PERFORM set_alv_headers USING lo_stock.

      lo_stock->get_display_settings(
        )->set_striped_pattern( abap_true ).

      lo_stock->get_display_settings(
        )->set_list_header(
        'SMARTPHARMA STOCK REPORT' ).

      lo_stock->display( ).

    CATCH cx_salv_msg INTO DATA(lx_stock).

      MESSAGE lx_stock->get_text( ) TYPE 'S'.

  ENDTRY.

ENDFORM.
